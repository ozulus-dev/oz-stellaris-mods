ring_noble_estates = {
	icon = "GFX_orbitalring_noble_estates"
	construction_days = 360
	starbase_type = orbital_ring
	category = sm_colony_cat

	potential = {
		exists = owner
		owner = {
			OR = {
				has_valid_civic = civic_aristocratic_elite
				has_valid_civic = civic_imperial_technocracy
			}
		}
		is_orbital_ring = yes
		planet = {
			NOR = {
				has_modifier = slave_colony
			}
		}
	}

	possible = {
		planet = {
			NOR = {
				has_building_construction = building_noble_estates
				has_building = building_noble_estates
			}
		}
	}

	abort_trigger = {
		exists = owner
		NOT = {
			owner = { OR = { has_civic = civic_aristocratic_elite has_civic = civic_imperial_technocracy } }
		}
	}

	destroy_trigger = {
		exists = owner
		NOT = {
			owner = { OR = { has_civic = civic_aristocratic_elite has_civic = civic_imperial_technocracy } }
		}
	}

	planet_modifier = {
		planet_housing_add = 200
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_worker_coop_empire = no
				is_gestalt = no
			}
		}
		job_politician_add = @building_static_jobs
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				is_worker_coop_empire = yes
			}
		}
		job_bureaucrat_add = @building_static_jobs
		mult = 1.5
	}

	resources = {
		category = starbase_buildings
		cost = {
			alloys = 200
			exotic_gases = 300
		}

		upkeep = {
			energy = 5
			exotic_gases = 2
		}
	}

	ai_build_at_chokepoint = yes
	ai_build_outside_chokepoint = yes

	ai_weight = {
		weight = 100
	}
}
