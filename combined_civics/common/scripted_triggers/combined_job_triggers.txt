politician_is_science_director = {
	OR = {
		owner? = {
			OR = {
				has_valid_civic = civic_technocracy
				has_valid_civic = civic_techno_aristocratic_guilds
			}
		}
		has_active_building = building_institute
		has_active_building = building_vultaum_reality_computer
		has_active_building = building_fe_lab_1
		has_active_building = building_fe_lab_2
	}
}

politician_is_noble = {
	owner? = {
		OR = {
			has_valid_civic = civic_aristocratic_elite
			has_valid_civic = civic_techno_aristocratic_guilds
			is_cyber_imperial = yes
		}
	}
}

politician_is_merchant = {
	OR = {
		owner? = {
			OR = {
				has_valid_civic = civic_merchant_guilds
				has_valid_civic = civic_techno_aristocratic_guilds
			}
		}
		has_active_building = building_galactic_stock_exchange
		has_active_building = building_fe_market_1
		has_active_building = building_fe_market_2
		AND = {
			has_branch_office = yes
			has_holding = {
				holding = building_imperial_concession_port
				owner = branch_office_owner
			}
		}
		AND = {
			has_branch_office = yes
			has_holding = {
				holding = building_corporate_embassy
				owner = branch_office_owner
			}
		}
		AND = {
			has_branch_office = yes
			has_holding = {
				holding = building_disinformation_center
				owner = branch_office_owner
			}
		}
	}
}
